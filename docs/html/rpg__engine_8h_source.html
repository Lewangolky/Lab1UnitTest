351template<typename T>
352class SkillTree {
353 unique_ptr<SkillTreeNode> root;
354public:
355 SkillTree() : root(nullptr) {}
356 SkillTree(Skill* s) { root = make_unique<SkillTreeNode>(s, nullptr); }
357
358 SkillTreeNode* getRoot() const { return root.get(); }
359
366 SkillTreeNode* insertUnder(const string& parentSkillName, Skill* s) {
367 if (!root) {
368 root = make_unique<SkillTreeNode>(s, nullptr);
369 return root.get();
370 }
371 SkillTreeNode* found = findNodeBySkillName(parentSkillName);
372 if (!found) return nullptr;
373 return found->addChild(s);
374 }
375
376 SkillTreeNode* findNodeBySkillName(const string& name) const {
377 if (!root) return nullptr;
378 SkillTreeNode* result = nullptr;
379 root->dfs([&](SkillTreeNode* node) {
380 if (node->getSkill() && node->getSkill()->getName() == name) result = node;
381 });
382 return result;
383 }
384
385 vector<string> descriptionsDFS() const {
386 vector<string> out;
387 if (!root) return out;
388 root->dfs([&](SkillTreeNode* node) {
389 if (node->getSkill()) out.push_back(node->getSkill()->description());
390 });
391 return out;
392 }
393
400 void generateRandom(Skill* (*skillFactory)(), int maxDepth = 3, int maxChildren = 3) {
401 root = make_unique<SkillTreeNode>(skillFactory(), nullptr);
402 default_random_engine rng((unsigned)chrono::high_resolution_clock::now().time_since_epoch().count());
403 queue<pair<SkillTreeNode*, int>> q;
404 q.push({ root.get(), 1 });
405 while (!q.empty()) {
406 auto [node, depth] = q.front(); q.pop();
407 if (depth >= maxDepth) continue;
408 uniform_int_distribution<int> distChildren(0, maxChildren);
409 int nc = distChildren(rng);
410 for (int i = 0;i < nc;i++) {
411 Skill* s = skillFactory();
412 SkillTreeNode* child = node->addChild(s);
413 q.push({ child, depth + 1 });
414 }
415 }
416 }

714class BattleSimulator {
715 Logger& logger;
716public:
717 BattleSimulator(Logger& log) : logger(log) {}
718
724 void simulate(Party& a, Party& b) {
725 logger.log("Battle starts between two parties!");
726 size_t turn = 0;
727 while (turn < 50) {
728 if (allDead(a)) { logger.log("Party A defeated!"); return; }
729 if (allDead(b)) { logger.log("Party B defeated!"); return; }
730
731 Character* ca = randomAlive(a);
732 Character* cb = randomAlive(b);
733 if (!ca || !cb) break;
734
735 if (turn % 2 == 0) {
736 ca->attack(*cb);
737 }
738 else {
739 cb->attack(*ca);
740 }
741 turn++;
742 }
743 logger.log("Battle ended after max turns.");
744 }
745
746private:
747 bool allDead(Party& p) {
748 for (size_t i = 0;i < p.size();++i) {
749 Character* c = p.getMember(i);
750 if (c && c->getHP() > 0) return false;
751 }
752 return true;
753 }
754 Character* randomAlive(Party& p) {
755 vector<Character*> alive;
756 for (size_t i = 0;i < p.size();++i) {
757 Character* c = p.getMember(i);
758 if (c && c->getHP() > 0) alive.push_back(c);
759 }
760 if (alive.empty()) return nullptr;
761 return alive[rand() % alive.size()];
762 }
