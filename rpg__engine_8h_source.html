266 void generateRandom(Skill* (*skillFactory)(), int maxDepth = 3, int maxChildren = 3) {
267 root = make_unique<SkillTreeNode>(skillFactory(), nullptr);
268 default_random_engine rng((unsigned)chrono::high_resolution_clock::now().time_since_epoch().count());
269 queue<pair<SkillTreeNode*, int>> q;
270 q.push({ root.get(), 1 });
271 while (!q.empty()) {
272 auto [node, depth] = q.front(); q.pop();
273 if (depth >= maxDepth) continue;
274 uniform_int_distribution<int> distChildren(0, maxChildren);
275 int nc = distChildren(rng);
276 for (int i = 0;i < nc;i++) {
277 Skill* s = skillFactory();
278 SkillTreeNode* child = node->addChild(s);
279 q.push({ child, depth + 1 });
280 }

503class BattleSimulator {
504 Logger& logger;
505public:
506 BattleSimulator(Logger& log) : logger(log) {}
507 void simulate(Party& a, Party& b) {
508 logger.log("Battle starts between two parties!");
509 size_t turn = 0;
510 while (turn < 50) {
511 if (allDead(a)) { logger.log("Party A defeated!"); return; }
512 if (allDead(b)) { logger.log("Party B defeated!"); return; }
513
514 Character* ca = randomAlive(a);
515 Character* cb = randomAlive(b);
516 if (!ca || !cb) break;
517
518 if (turn % 2 == 0) {
519 ca->attack(*cb);
520 }
521 else {
522 cb->attack(*ca);
523 }
524 turn++;
525 }
526 logger.log("Battle ended after max turns.");
527 }
528
529private:
530 bool allDead(Party& p) {
531 for (size_t i = 0;i < p.size();++i) {
532 Character* c = p.getMember(i);
533 if (c && c->getHP() > 0) return false;
534 }
535 return true;
536 }
537 Character* randomAlive(Party& p) {
538 vector<Character*> alive;
539 for (size_t i = 0;i < p.size();++i) {
540 Character* c = p.getMember(i);
541 if (c && c->getHP() > 0) alive.push_back(c);
542 }
543 if (alive.empty()) return nullptr;
544 return alive[rand() % alive.size()];
545 }
